import { rangoDeDias } from './rango-de-dias';

describe('Reproducción de bugs reales en producción (casos de MongoDB a medianoche UTC)', () => {
  it('Caso 1 — Licencia médica TT-2026-00471 (desfase de días dentro del mismo mes)', () => {
    expect(
      rangoDeDias('2026-08-17T00:00:00.000Z', '2026-08-28T00:00:00.000Z'),
    ).toEqual({
      etiqueta: 'del 17 al 28 de agosto',
      dias: 12,
    });
  });

  it('Caso 2 — Feriado legal de septiembre (inicio de mes retrocede a agosto)', () => {
    expect(
      rangoDeDias('2026-09-01T00:00:00.000Z', '2026-09-15T00:00:00.000Z'),
    ).toEqual({
      etiqueta: 'del 1 al 15 de septiembre',
      dias: 15,
    });
  });

      etiqueta: 'el 1 de enero',

});

 *   → { etiqueta: 'del 17 al 28 de agosto', dias: 12 }
 */

const MESES = [
  'enero', 'febrero', 'marzo', 'abril', 'mayo', 'junio',
  'julio', 'agosto', 'septiembre', 'octubre', 'noviembre', 'diciembre',
];

const UN_DIA = 24 * 60 * 60 * 1000;

export interface RangoDeDias {
  /** «del 17 al 28 de agosto», o «el 17 de agosto» si es un solo día. */
  etiqueta: string;
  /** Días corridos, ambos extremos incluidos. */
  dias: number;
}

/** El día del mes de la fecha calendario (UTC). */
function diaDe(f: Date): number {
  return f.getUTCDate();
}

/** El mes (0-11) de la fecha calendario (UTC). */
function mesDe(f: Date): number {
  return f.getUTCMonth();
}

/** El año de la fecha calendario (UTC). */
function anioDe(f: Date): number {
  return f.getUTCFullYear();
}

export function rangoDeDias(desde: string | Date, hasta: string | Date): RangoDeDias {
  const a = desde instanceof Date ? desde : new Date(desde);
  const b = hasta instanceof Date ? hasta : new Date(hasta);

  if (Number.isNaN(a.getTime()) || Number.isNaN(b.getTime())) {
    throw new Error('Fecha inválida');
  }
  if (b.getTime() < a.getTime()) {
    throw new Error('El término no puede ser anterior al inicio');
  }

  // Los días corridos se cuentan sobre el instante, sin pasar por la zona:
  // dos fechas de calendario consecutivas siempre difieren en 24 h exactas.
  const dias = Math.round((b.getTime() - a.getTime()) / UN_DIA) + 1;

  const dA = diaDe(a);
  const dB = diaDe(b);
  const mA = mesDe(a);
  const mB = mesDe(b);
  const yA = anioDe(a);
  const yB = anioDe(b);

  if (dias === 1) {
    return { etiqueta: `el ${dA} de ${MESES[mA]}`, dias };
  }

  if (yA !== yB) {
    return {
      etiqueta: `del ${dA} de ${MESES[mA]} de ${yA} al ${dB} de ${MESES[mB]} de ${yB}`,
      dias,
    };
  }

  if (mA !== mB) {
    return {
      etiqueta: `del ${dA} de ${MESES[mA]} al ${dB} de ${MESES[mB]}`,
      dias,
    };
  }

  return { etiqueta: `del ${dA} al ${dB} de ${MESES[mA]}`, dias };
}

Objetos Date creados en horas nocturnas locales (desfase inverso):
Si un consumidor externo instancia un objeto Date en horario local chileno tarde en la noche (por ejemplo, new Date(2026, 7, 17, 22, 0) a las 22:00 del 17 de agosto), en UTC ese timestamp se serializa a las 02:00 del día siguiente (2026-08-18T02:00:00.000Z). Como nuestra función extrae componentes con getUTCDate(), leerá el día 18 en vez del 17. La función asume que los objetos Date provienen normalizados a medianoche UTC (como hace MongoDB) o como cadenas ISO sin hora.

Inconsistencia entre etiqueta y días si se pasan timestamps con horas dispares:
Si se ingresa un desde a las 23:30 de un día y un hasta a las 00:30 del día siguiente (solo 1 hora de diferencia real), el cálculo (b - a) / UN_DIA da 0.041, que redondea a 0 + 1 = 1 día. Sin embargo, las fechas calendario cruzan dos días distintos (17 y 18), generando una etiqueta del tipo «del 17 al 18 de agosto» con dias: 1. Hay un desacople si las fechas traen horas arbitrarias.

Formatos de fecha regionales no normalizados:
Cadenas en formatos como '17/08/2026' o '17-08-2026' no son parseadas de forma determinista por el constructor nativo de Date en todos los motores de JavaScript, arrojando Fecha inválida o invirtiendo mes por día.
